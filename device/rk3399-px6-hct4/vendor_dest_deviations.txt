# Vendor destinations deliberately NOT restored - neither the staged blob nor an Android 14 module.
# Accepted by the owner on Nov 6 (v1). tools/vendor_dropped_check.py
# reports them but never fails on them, so the verification gate stays meaningful for everything else.
# The block added on Nov 16 (**D52**) is declared by engineering, not accepted by the owner yet; it is
# the same mechanism, so the gate keeps reporting it and keeps passing for everything else.
# Paths are partition-relative (`bin/hw/rild` for /vendor/bin/hw/rild). Comments give the reason.

# ---- no cellular modem --------------------------------------------------------------
# This unit has no radio: phone calls arrive over Bluetooth (the vendor's MTC/CarPlay stack), there
# is no SIM slot and no modem device in /dev. Android 14's RIL daemon would start, find nothing to
# talk to and loop. hardware/ril's Make rules install these paths, so the stock blobs are dropped
# (vendor_dest_skip_extra.txt) and the daemon is not requested in PRODUCT_PACKAGES.
bin/hw/rild
etc/init/rild.rc
lib/libril.so
lib/libreference-ril.so
lib64/libril.so
lib64/libreference-ril.so

# ---- no boot->recovery flash-recovery trigger --------------------------------------
# `vendor_flash_recovery.rc` runs `vendor_flash_recovery` early in boot, which re-flashes the
# recovery partition from /vendor/recovery-from-boot.p. That file is not staged (it is an Android 11
# payload for the old GPT layout), so the trigger would act on a file that does not exist. Recovery
# is written explicitly during the first flash (docs/flash.md) and by the Updater.
etc/init/vendor_flash_recovery.rc

# ---- no HIDL wifi service (**D21**) ----------------------------------------------
# Android 14's WiFi framework dropped HIDL entirely (`grep -rl 'android.hardware.wifi@'
# frameworks/opt/net/wifi/service/java/ -> 0 files`), so `android.hardware.wifi@1.0-service` would
# start, register, and never be spoken to. WiFi comes from Android 14's own AIDL service instead
# (`android.hardware.wifi-service` + `libwifi-hal` -> `libwifi-hal-rtk`, device.mk and BoardConfig.mk).
# `librkwifi-ctrl.so` only served that service (`check_wifi_chip_type_string`, `get_wifi_device_id`);
# the module loading it used to do is hu_insmod.sh's job.
# `lib/libwifi-hal.so` (the blob's 32-bit glue) is NOT a deviation any more: **D23** gives Android 14's
# libwifi-hal `compile_multilib: "both"`, so the build installs that destination itself and
# `bin/rockchip.drmservice` resolves `wifi_load_driver` there. The blob copy is dropped because the
# build owns the path (vendor_dest_skip_extra.txt), not because it is lost.
bin/hw/android.hardware.wifi@1.0-service
etc/init/android.hardware.wifi@1.0-service.rc
lib64/librkwifi-ctrl.so
lib/librkwifi-ctrl.so

# ---- no Gatekeeper at all (**D52**) --------------------------------------------
# Android 14 removed Gatekeeper (keystore2 attestation goes through KeyMint), so no module installs
# the service or the impl. The Rockchip stack cannot be carried over either: the DT_NEEDED walk of
# the kept blobs shows `gatekeeper.rk30board.so` and `librkgatekeeper.so` need `libgatekeeper.so`,
# and `nm -D --undefined-only` shows 8 `gatekeeper::` symbols (EnrollRequest/VerifyRequest/
# EnrollResponse/VerifyResponse ctors+dtors) that no file in this build tree, in the Android 10
# reference firmware or in the Android 11 reference firmware defines. `libgatekeeper.so` is not in
# the stock Android 10 system or vendor partition either - Gatekeeper was already non-functional on
# the OEM firmware. `libteec.so` (Trusty TEE client) IS staged, so the TEE layer alone would load;
# the missing libgatekeeper.so is what decides it. Shipping nine files that cannot be dlopened is
# worse than declaring the gap, so these blobs are dropped as well.
etc/init/android.hardware.gatekeeper@1.0-service.rc
lib/hw/android.hardware.gatekeeper@1.0-impl.so
lib64/hw/android.hardware.gatekeeper@1.0-impl.so

# ---- libwpa_client: nothing links it (**D52**) --------------------------------

# ---- no HIDL light HAL (**D52**) ----------------------------------------------
# Android 14's lights are AIDL: frameworks/base/core/java/android/hardware/lights/SystemLightsManager.java
# and frameworks/base/services/core/java/com/android/server/lights/LightsService.java, and a grep for
# `light@2` in frameworks/native and frameworks/base finds no service code (only tests, jni, Watchdog
# and VintfObject). Android 14's light@2.0 default impl is a passthrough over the legacy `lights` HAL,
# and the legacy lookup never opens the blob this unit ships: it tries `lights.<ro.hardware>.so`
# (lights.rk30board.so - absent) and then `lights.default.so` (absent), while the staged file is named
# `lights.rk3399.so`. Starting the service would fail at impl load and init would restart it, so it is
# declared rather than restored. If LED control ever becomes a criterion, the fix is a provider named
# lights.default.so (or a ro.hardware.lights override), not this blob.
etc/init/android.hardware.light@2.0-service.rc
lib/hw/android.hardware.light@2.0-impl.so
lib64/hw/android.hardware.light@2.0-impl.so

# ---- no drm@1.0 service/impl (**D52**) ----------------------------------------
# The kept Android 10 DRM services are the 1.2 ones (android.hardware.drm@1.2-service.clearkey and
# .widevine, both restored and both registered in their own init scripts), and Android 14's
# libmediadrm still carries the HIDL path (frameworks/av/drm/libmediadrm/DrmHalHidl.cpp and
# CryptoHalHidl.cpp), so a 1.2 factory - which casts down to 1.1 and 1.0 instances - already covers
# every version the framework asks for. Android 14's drm@1.0 default impl is a passthrough over the
# legacy `drm` HAL, which the HU does not ship separately; adding a second service would compete for
# the same interface names without adding a plugin.
etc/init/android.hardware.drm@1.0-service.rc
lib/hw/android.hardware.drm@1.0-impl.so
lib64/hw/android.hardware.drm@1.0-impl.so

# ---- no cas@1.1 service rc (**D52**) ------------------------------------------
# Android 14's CAS service is `android.hardware.cas@1.2-service` (requested in device.mk), which
# installs its own init script and registers ICasFactory at 1.2 - castable to 1.1 and 1.0 - so the
# OEM 1.1 service would be a second provider of the same interface. The CAS plugin itself
# (lib*/mediacas/libclearkeycasplugin.so) is build-owned (Android 14's own copy); note that the OEM

# so CAS was already dead there.
etc/init/android.hardware.cas@1.1-service.rc


# ---- no Android 10 health service blob (**D52b**) -----------------------------
# `tools/vendor_abi_check.py` measured that the kept blob cannot load on Android 14:
# `bin/hw/android.hardware.health@2.0-service` needs
# `_ZN7android7String8C1ENS0_13StaticLinkageE`, Android 10-era libutils' String8 StaticLinkage
# constructor, which Android 14's libutils does not export (String8's StaticLinkage API became
# inline). Android 14 defines no install rule for this service path (`vendor_staging.mk` has a copy
# entry for it, `vendor_build_wins.txt` has none), so the blob really runs - init would start it, see
# the unresolved symbol at dlopen/ld.so time, and restart it on backoff, while battery info is
# served anyway by Android 14's own `android.hardware.health@2.1-service` +
# `android.hardware.health@2.1-impl` (requested in device.mk; the image carries
# `bin/hw/android.hardware.health@2.1-service` and `lib*/hw/android.hardware.health@2.0-impl-2.1.so`,
# which is Android 14's 2.0-compatible impl). Declaring the gap also stops the OEM manifest's
# `health@2.0` declaration from pointing at a service that cannot start.
bin/hw/android.hardware.health@2.0-service
etc/init/android.hardware.health@2.0-service.rc

# ---- no Rockchip OMX VPU components (**D52b**) --------------------------------
# These five components cannot load on Android 14: `lib*/libomxvpu_enc.so` and
# `lib*/libomxvpu_dec.so` need `_ZN7android19GraphicBufferMapper4lockEPK13native_handlejRKNS_4RectEPPvPiS9_`
# and `_ZN7android7String8...`-era `GraphicBufferMapper::unlock`, i.e. the legacy gralloc1 lock/unlock
# exports of Android 10's libui. Android 14's libui has no such symbols (buffer locking moved into the
# mapper interface, and `libui.so` is a public library loaded from /system - shadowing it with a
# /vendor copy would give one process two libui copies, which is worse than losing a codec).
# `lib*/libOMX_Core.so`, `lib*/libRkOMX_Resourcemanager.so` and `lib*/libgralloc_priv_omx.so` exist
# only to serve those two plugins: a DT_NEEDED scan over every staged, non-dropped vendor ELF finds
# libOMX_Core.so and libRkOMX_Resourcemanager.so used solely by lib*/libOMX_Core.so + the plugins, and
# libgralloc_priv_omx.so solely by the plugins. Nothing in Android 14 loads them either: the OMX
# service this ROM ships is Android 14's own `android.hardware.media.omx@1.0-service`, whose component
# table is AOSP's (soft-OMX), and the plugins are dlopened by name rather than linked, so the
# DT_NEEDED walk never saw them.
# Consequence, stated plainly: hardware VPU encode/decode through OMX is lost. Software codecs
# (libstagefright_softomx + the requested encoders/decoders) and MediaCodec's AIDL/C2 path replace it,
# and the VPU is still usable for JPEG encode: `lib*/libjpeghwenc.so` and
# `lib*/hw/camera.rk30board.so` link `lib*/libvpu.so` directly (measured: 6 consumers), which is why
# `lib*/libvpu.so` and `lib*/librk_vpuapi.so` stay staged.
lib/libomxvpu_enc.so
lib64/libomxvpu_enc.so
lib/libomxvpu_dec.so
lib64/libomxvpu_dec.so
lib/libOMX_Core.so
lib64/libOMX_Core.so
lib/libRkOMX_Resourcemanager.so
lib64/libRkOMX_Resourcemanager.so
lib/libgralloc_priv_omx.so
lib64/libgralloc_priv_omx.so
