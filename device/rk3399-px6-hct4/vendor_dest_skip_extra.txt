# Vendor destinations to leave out of PRODUCT_COPY_FILES because the *build* installs that path.
# tools/vendor_conflicts.py finds the ones installed by Soong modules (it reads
# out/soong/installs-lineage_rk3399.mk). Legacy Make modules install paths too, but nothing
# enumerates them statically - ckati only says so when the copy list already conflicts, and
# scripts/package.sh records the answer here so a fresh build host converges in one pass.
# Paths are partition-relative, i.e. `bin/hw/rild` for /vendor/bin/hw/rild. One per line; '#' is
# a comment. scripts/prebuilt_vendor.sh merges these into vendor_build_wins.txt before generating.

# ---- revealed by ckati in pass los-pack16..pack18 (legacy Make install rules) ----
# All of these are Android 14's own build of the same component, so the build's file wins.

# The reference RIL daemon and its libraries (ckati: "previously defined at
# build/make/core/base_rules.mk:497"). hardware/rockchip_ril builds rild/libril/libreference-ril
# for Android 14; the HU's radio HAL implementation stays the vendor blob.
bin/hw/rild
etc/init/rild.rc
lib/libril.so
lib/libreference-ril.so
lib64/libril.so
lib64/libreference-ril.so

# The legacy gralloc stub module (hardware/libhardware/modules/gralloc), installed for both
# architectures. This unit's real graphics HAL is the staged midgard allocator/mapper blobs, so
# dropping the stub loses nothing.
lib/hw/gralloc.default.so
lib64/hw/gralloc.default.so

# The fs_config metadata. The stock vendor ships two empty files here; Android 14 generates real
# ones from the vendor file-contexts table, so the blobs must not be copied.
etc/fs_config_dirs
etc/fs_config_files

# The notices blob, generated by build/make/core/os_licensing.mk:44 from every module in the
# product - the stock Android 11 one lists Android 11 libraries.
etc/NOTICE.xml.gz

# ---- WiFi control plane (D21): Android 14 builds these itself ----
# external/wpa_supplicant_8/wpa_supplicant/Android.mk installs the AIDL supplicant service as
# vendor/bin/hw/wpa_supplicant (LOCAL_MODULE_RELATIVE_PATH := hw), carrying
# aidl/android.hardware.wifi.supplicant.xml and supplicant-service.rc; hostapd's Android.mk installs
# vendor/bin/hw/hostapd, vendor/etc/init/hostapd.android.rc and android.hardware.wifi.hostapd.xml.
# The staged blobs of the same names cannot load here anyway: the blob `wpa_supplicant` needs
# `sk_dup`, which Android 14's OpenSSL 3 no longer exports (tools/vendor_abi_check.py reports it).
bin/hw/wpa_supplicant
bin/hw/hostapd
etc/init/hostapd.android.rc

# Same Make-built pair for the client tools: wpa_cli and libwpa_client are built from
# external/wpa_supplicant_8 (wpa_supplicant/Android.mk:1912 and :2043), so the stock copies collide
# with them (ckati: "overriding commands for target .../vendor/bin/wpa_cli, previously defined at
# build/make/core/base_rules.mk:497"). Android 14's copy is the ABI its own supplicant talks to.
bin/wpa_cli
lib/libwpa_client.so
lib64/libwpa_client.so

# ---- D23: libwifi-hal's 32-bit variant ----
# `lib/libwifi-hal.so` is the Android 11 blob's 32-bit WiFi HAL glue. patches/frameworks-opt-net-
# wifi-libwifi-hal-32bit.patch gives Android 14's own libwifi-hal `compile_multilib: "both"`, so the
# build installs /vendor/lib/libwifi-hal.so itself (the 64-bit copy already won this path through
# vendor_build_wins.txt). The blob must not come back: its own copy has 32 unresolved symbols (no
# 32-bit libnl provider in the staged tree, libnl 3.8 ABI drift), and `bin/rockchip.drmservice` - a
# 32-bit ELF - resolves `wifi_load_driver` against the built copy.
lib/libwifi-hal.so

# ---- D23: usb_dongle is rebuilt in-tree ----
# device/rockchip/rk3399-px6-hct4/usb_dongle/Android.bp builds `usb_dongle` and installs
# /vendor/bin/usb_dongle plus /vendor/etc/init/usb_dongle.rc (init_rc). Same destinations as the
# blob, so the blob copies are dropped; the blob cannot load here (five legacy libsysutils symbols,
# see vendor_dest_deviations.txt for why it is not declared as a gap).
# tools/vendor_conflicts.py classifies a path installed by a *device-tree* module as "blob must win"
# (that heuristic exists to stop an AOSP default HAL from shadowing a Rockchip HAL). Here it is the
# opposite: the device-tree module IS the blob's own source rebuilt for Android 14, so this file is
# the override that drops the blob - the final conflict scan reports "no collisions".
bin/usb_dongle
etc/init/usb_dongle.rc
